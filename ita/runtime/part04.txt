<script>
(function () {
  var revealEls = document.querySelectorAll('.reveal');

  function revealAll() {
    for (var i = 0; i < revealEls.length; i++) revealEls[i].classList.add('is-in');
  }

  if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (e) {
        if (e.isIntersecting) { e.target.classList.add('is-in'); io.unobserve(e.target); }
      });
    }, { rootMargin: '0px 0px -6% 0px', threshold: 0.02 });
    Array.prototype.forEach.call(revealEls, function (el) { io.observe(el); });
  } else {
    revealAll();
  }
  // Failsafe: never leave a reader staring at a blank page.
  setTimeout(revealAll, 1600);

  // Click-to-play video: nothing loads from YouTube until asked.
  Array.prototype.forEach.call(document.querySelectorAll('.video'), function (fig) {
    var btn = fig.querySelector('.video__play');
    if (!btn) return;
    btn.addEventListener('click', function () {
      var id = fig.getAttribute('data-yt');
      var fb = fig.getAttribute('data-fb');
      if (!id && !fb) return;
      // Embeds need a real http(s) origin. Opened straight off disk there isn't
      // one, and YouTube answers with Error 153 - so send people to the source.
      if (location.protocol === 'file:') {
        window.open(id ? 'https://www.youtube.com/watch?v=' + id : fb, '_blank', 'noopener');
        return;
      }
      var frame = document.createElement('iframe');
      frame.className = 'video__frame';
      frame.src = id
        ? 'https://www.youtube-nocookie.com/embed/' + id + '?autoplay=1&rel=0'
        : 'https://www.facebook.com/plugins/video.php?href=' + encodeURIComponent(fb) +
          '&show_text=false&autoplay=true';
      frame.title = btn.getAttribute('aria-label') || 'Video';
      frame.allow = 'accelerometer; autoplay; encrypted-media; picture-in-picture';
      frame.allowFullscreen = true;
      fig.replaceChild(frame, btn);
    });
  });

  // Video links open an in-page player rather than leaving the site.
  var lb = document.getElementById('lightbox');
  var lbBox = lb.querySelector('.lb__box');
  var lbClose = lb.querySelector('.lb__close');
  var lastFocus = null;

  function ytId(href) {
    var m = href.match(/(?:[?&]v=|youtu\.be\/|\/embed\/)([\w-]{6,})/);
    return m ? m[1] : null;
  }

  function openPlayer(src, title) {
    lbBox.innerHTML = '';
    var f = document.createElement('iframe');
    f.className = 'lb__frame';
    f.src = src;
    f.title = title || 'Video';
    f.allow = 'accelerometer; autoplay; encrypted-media; picture-in-picture';
    f.allowFullscreen = true;
    lbBox.appendChild(f);
    lb.hidden = false;
    document.body.style.overflow = 'hidden';
    lbClose.focus();
  }

  function closePlayer() {
    lb.hidden = true;
    lbBox.innerHTML = '';
    document.body.style.overflow = '';
    if (lastFocus && lastFocus.focus) lastFocus.focus();
  }

  document.addEventListener('click', function (e) {
    if (location.protocol === 'file:') return; // no origin to embed from; follow the link
    if (e.metaKey || e.ctrlKey || e.shiftKey || e.button === 1) return; // let people open a new tab
    var a = e.target.closest && e.target.closest('a[href]');
    if (!a) return;
    var h = a.getAttribute('href') || '';
    var label = (a.textContent || '').trim().slice(0, 90);

    if (/youtube\.com|youtu\.be/.test(h)) {
      var id = ytId(h);
      if (!id) return;
      e.preventDefault();
      lastFocus = a;
      openPlayer('https://www.youtube-nocookie.com/embed/' + id + '?autoplay=1&rel=0', label);
    } else if (/facebook\.com\/(watch|[^\/]+\/videos)/.test(h)) {
      e.preventDefault();
      lastFocus = a;
      openPlayer('https://www.facebook.com/plugins/video.php?href=' + encodeURIComponent(h) +
                 '&show_text=false&autoplay=true', label);
    }
  });

  lb.addEventListener('click', function (e) {
    if (e.target === lb || e.target === lbClose) closePlayer();
  });
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && !lb.hidden) closePlayer();
  });

  var bar = document.getElementById('topbar');
  var prog = document.getElementById('progress');
  var links = Array.prototype.slice.call(document.querySelectorAll('.topbar__nav a'));
  var targets = links.map(function (l) { return document.querySelector(l.getAttribute('href')); });

  function onScroll() {
    var y = window.scrollY || document.documentElement.scrollTop;

    if (y > window.innerHeight * 0.5) bar.classList.add('is-visible');
    else bar.classList.remove('is-visible');

    var max = document.documentElement.scrollHeight - window.innerHeight;
    prog.style.transform = 'scaleX(' + (max > 0 ? Math.min(y / max, 1) : 0) + ')';

    var active = -1;
    for (var i = 0; i < targets.length; i++) {
      if (targets[i] && targets[i].getBoundingClientRect().top <= window.innerHeight * 0.35) active = i;
    }
    links.forEach(function (l, i) {
      if (i === active) l.classList.add('is-active');
      else l.classList.remove('is-active');
    });
  }

  window.addEventListener('scroll', onScroll, { passive: true });
  window.addEventListener('resize', onScroll);
  onScroll();
})();
